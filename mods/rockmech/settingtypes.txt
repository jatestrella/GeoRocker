[Lithology and stress]

# Ground surface elevation y0 (m). Depth h = y0 - y. 1 node = 1 m.
# Also controls the depth layering of lithologies (new chunks only).
rockmech_surface_y (Surface elevation y0) int 0 -31000 31000

# Lateral earth pressure coefficient K0: sigma_h = K0 * sigma_v.
rockmech_k0 (K0 = sigma_h / sigma_v) float 0.5 0.0 10.0

# Major horizontal principal stress: sigma_H = K_H * sigma_v + sigma_H0. Empty = use K0.
rockmech_k_hmax (K_H for sigma_H) string

# Minor horizontal principal stress: sigma_h = K_h * sigma_v + sigma_h0. Empty = use K0.
rockmech_k_hmin (K_h for sigma_h) string

# Tectonic (depth-independent) part of sigma_H (MPa).
rockmech_sigma_hmax0 (sigma_H0 tectonic term) float 0 0 1000

# Tectonic (depth-independent) part of sigma_h (MPa).
rockmech_sigma_hmin0 (sigma_h0 tectonic term) float 0 0 1000

# Azimuth of sigma_H (degrees clockwise from north). 90 = east-west.
rockmech_sigma_hmax_azimuth (sigma_H azimuth) float 0 0 360

# Replace mapgen stone with rockmech lithologies by depth layer (new chunks only).
rockmech_replace_stone (Replace mapgen stone) bool true

# Remove soil (dirt, sand, gravel, clay, snow) during map generation to expose bedrock.
# Usually set per world in <world>/rockmech.conf rather than globally.
rockmech_strip_soil (Strip soil) bool false

# Replace every other natural solid node (ores, permafrost, ice, biome stones, ...) with the
# lithology at that position during map generation, so only rockmech lithologies remain.
# Usually set per world in <world>/rockmech.conf.
rockmech_only_lithologies (Only lithologies) bool false

# Lithology distribution. layers: by depth (sandstone, shale, granite downward).
# zones: vertical zones along x, each zone_width wide (shale around x = 0, granite to the east,
# sandstone to the west, repeating). Usually set per world in <world>/rockmech.conf.
rockmech_lith_mode (Lithology distribution) enum layers layers,zones

# Width (m) of each lithology zone in zones mode.
rockmech_zone_width (Lithology zone width) int 80 4 10000

[Joint sets]

# Joint plane normals are snapped to an integer lattice of this density so that joint
# traces connect exactly between blocks. Larger = closer to the input orientation
# (about 1 degree at 8) but more texture variants. The effective orientation is shown
# by the compass. Changing it changes the joint geometry of the world.
rockmech_joint_lattice (Joint orientation lattice) int 8 2 16

# Override joint set 1: "dip,dipdir,spacing,persistence,phi"
# (degrees, degrees clockwise from north, m >= 1, 0-1, degrees).
# Empty = use the default in config.lua.
rockmech_joint_set_1 (Joint set 1 override) string

# Override joint set 2: "dip,dipdir,spacing,persistence,phi". Empty = default.
rockmech_joint_set_2 (Joint set 2 override) string

# Override joint set 3: "dip,dipdir,spacing,persistence,phi". Empty = default.
rockmech_joint_set_3 (Joint set 3 override) string

[Deviatoric failure]

# Check exposed rock for deviatoric failure (sigma1 - sigma3 >= UCS * scale) after excavation.
rockmech_failure_enabled (Enable deviatoric failure) bool true

# Radius (m) of the region checked around each excavation.
rockmech_failure_radius (Check radius) int 12 1 32

# Half-width w of the horizontal load-spreading window (2w+1)^2.
# sigma_y = (sum of column loads in the window, incl. roof loads over openings) / (solid cells in window).
rockmech_failure_window (Load-spreading window half-width) int 4 1 16

# Unit weight (kN/m3) of non-lithology solid nodes (dirt, ores, ...) in the column load.
rockmech_failure_gamma_other (Unit weight of other solids) float 20 0 50

# Height (m) scanned upward for the column load; rock beyond is assumed continuous up to y0.
rockmech_failure_overburden_scan (Overburden scan height) int 64 0 512

# Initial in-situ stress (HUD / info page / compass): maximum height scanned upward to find the ground surface (m).
# Voids below the surface are counted as rock (pre-excavation state); beyond this height rock is assumed to continue to surface_y.
rockmech_failure_initial_scan_max (Initial stress scan height) int 512 16 4096

# Failure threshold = UCS * scale. 1 = real strength; smaller values make failure visible at shallow depth.
rockmech_failure_strength_scale (Strength scale) float 1.0 0.001 10

# Maximum number of cascade rounds after a failure.
rockmech_failure_max_cascade (Max cascade rounds) int 5 0 50

# Delay between cascade rounds (s).
rockmech_failure_cascade_delay (Cascade delay) float 0.6 0 10

# Log a warning when one check takes longer than this (ms).
rockmech_failure_budget_ms (Time budget per check) float 5 0.1 1000

[Rockfall]

# Falling rock becomes a tumbling entity (bounce, roll, stop) instead of dropping straight down.
rockmech_rockfall_enabled (Enable tumbling rockfall) bool true

# Maximum number of tumbling rocks at once; extra rocks fall straight down.
rockmech_rockfall_max_active (Max tumbling rocks) int 64 0 1000

# Normal restitution Rn of non-lithology surfaces (dirt, rubble, ...). Lithologies use config.lua.
rockmech_rockfall_rn (Normal restitution of other surfaces) float 0.30 0 1

# Tangential restitution Rt of non-lithology surfaces.
rockmech_rockfall_rt (Tangential restitution of other surfaces) float 0.75 0 1

# Rolling resistance (tan of rolling friction angle) of non-lithology surfaces.
rockmech_rockfall_mu_roll (Rolling resistance of other surfaces) float 0.55 0 5

# Normal impact speed (m/s) above which a contact is a bounce; below it the rock rolls.
rockmech_rockfall_impact_speed (Impact speed threshold) float 2.0 0 20

# A rock in contact with the ground slower than this (m/s) for 0.5 s stops and becomes a node.
rockmech_rockfall_stop_speed (Stop speed) float 0.4 0 10

# Maximum tumbling time (s).
rockmech_rockfall_max_time (Max tumbling time) float 20 1 300

[Angle of repose]

# Rubble steeper than the angle of repose moves (slides or tumbles). Intact rock is not affected.
# Can also be toggled in game.
rockmech_slope_enabled (Enable rubble angle-of-repose sliding) bool true

# Angle of repose of rubble (degrees). Angular rock debris is typically 34-40.
rockmech_slope_angle (Rubble angle of repose) float 35 1 89

[Support]

# When something is removed, rock groups no longer connected to the rock mass
# (and not resting on anything) fall as vanilla falling nodes.
rockmech_support_enabled (Enable support check) bool true

# Search radius (m). Reaching the edge of this region counts as connected to the rock mass.
rockmech_support_radius (Support search radius) int 12 2 32

# Unsupported groups larger than this are conservatively treated as stable.
rockmech_support_max_voxels (Max falling group size) int 2000 1 100000

# Unsupported groups fall as one rigid block (vertical translation, no rotation).
# Off: each node falls separately (vanilla falling nodes).
rockmech_support_rigid (Rigid block falling) bool true

# Groups larger than this fall node by node instead of as one block.
rockmech_support_rigid_max (Max rigid block size) int 128 1 2000

# Log a warning when one support check takes longer than this (ms).
rockmech_support_budget_ms (Support check time budget) float 5 0.1 1000

[Blasting]

# TNT explosions fragment rock: within r_out = R * (ucs_ref / UCS)^(1/3) of the blast centre rock
# turns into rubble; within inner * r_out it is thrown out as rubble items.
# Requires the tnt mod. Can also be toggled in game.
rockmech_blast_enabled (Enable blast fragmentation) bool true

# Reference UCS (MPa) whose fragmentation radius equals the blast radius.
rockmech_blast_ucs_ref (Blast reference UCS) float 60 0.1 1000

# Fraction of the fragmentation radius inside which rock is thrown out.
rockmech_blast_inner (Blast ejection fraction) float 0.5 0 1

[Overhang]

# Overhanging rock (cantilevers, thin roofs) breaks at its root and falls when its distance from
# a supported rock column exceeds L_max = sqrt(sigma_t * scale * t / (3 * gamma)),
# where t is the local rock thickness. Can also be toggled in game.
rockmech_overhang_enabled (Enable overhang breaking) bool true

# Tensile strength scale (1 = real strength from config.lua).
rockmech_overhang_scale (Overhang tensile strength scale) float 1.0 0.001 100

# Tensile strength (MPa) of non-lithology bonded nodes (dirt, ores, ...).
rockmech_overhang_tensile_other (Tensile strength of other nodes) float 2.0 0 100

# Check radius (m). Overhangs reaching the edge of this region are left alone (conservative).
rockmech_overhang_radius (Overhang check radius) int 16 4 48

# Overhangs larger than this many nodes are left alone (conservative).
rockmech_overhang_max_voxels (Max overhang size) int 2000 1 100000

[Fall breakage]

# Intact rock that falls may break into rubble on landing: p = 1 - exp(-h / h_ref).
# h_ref of each lithology is set in config.lua (break_height).
rockmech_fallbreak_enabled (Break falling rock into rubble) bool true

# h_ref (m) for non-lithology rock such as default:stone.
rockmech_fallbreak_break_height (Break height of other rock) float 4.0 0.01 1000

[Joint textures]

# Place joint texture variants (trace lines on block edges, colored joint faces)
# during map generation and /convert_rocks. Texture changes need a world restart.
rockmech_jointtex_enabled (Joint textures) bool true

# plane: draw the true trace of each joint plane across block faces (inclined lines).
# edges: draw lines along block edges where the neighboring face is a joint, and tint joint faces.
rockmech_jointtex_style (Joint texture style) enum plane plane,edges

# plane style: maximum number of joint sets drawn in one block.
rockmech_jointtex_max_lines (Max joint traces per block) int 2 1 3

# plane style: trace texture resolution in pixels.
rockmech_jointtex_plane_res (Trace texture resolution) int 32 16 64

# plane style: trace width in pixels (at the trace texture resolution).
rockmech_jointtex_plane_width (Trace width) float 1.5 0.5 8

# Color of joint trace lines (#RRGGBB).
rockmech_jointtex_line_color (Joint trace color) string #000000

# Opacity of joint trace lines (0-255).
rockmech_jointtex_line_alpha (Joint trace opacity) int 170 0 255

# Width of joint trace lines in pixels (textures are 16x16).
rockmech_jointtex_line_width (Joint trace width) int 1 1 8

# Tint of exposed joint faces (#RRGGBB).
rockmech_jointtex_face_color (Joint face tint) string #8a5a2a

# Strength of the joint face tint (0-255).
rockmech_jointtex_face_ratio (Joint face tint strength) int 110 0 255
