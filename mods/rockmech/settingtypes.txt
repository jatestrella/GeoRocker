[Lithology and stress]

# Ground surface elevation y0 (m). Depth h = y0 - y. 1 node = 1 m.
# Also controls the depth layering of lithologies (new chunks only).
rockmech_surface_y (Surface elevation y0) int 0 -31000 31000

# Lateral earth pressure coefficient K0: sigma_h = K0 * sigma_v.
rockmech_k0 (K0 = sigma_h / sigma_v) float 0.5 0.0 10.0

# Major horizontal principal stress: sigma_H = K_H * sigma_v + sigma_H0. Empty = use K0.
rockmech_k_hmax (K_H for sigma_H) string

# Minor horizontal principal stress: sigma_h = K_h * sigma_v + sigma_h0. Empty = use K0.
rockmech_k_hmin (K_h for sigma_h) string

# Tectonic (depth-independent) part of sigma_H (MPa).
rockmech_sigma_hmax0 (sigma_H0 tectonic term) float 0 0 1000

# Tectonic (depth-independent) part of sigma_h (MPa).
rockmech_sigma_hmin0 (sigma_h0 tectonic term) float 0 0 1000

# Azimuth of sigma_H (degrees clockwise from north). 90 = east-west.
rockmech_sigma_hmax_azimuth (sigma_H azimuth) float 0 0 360

# Replace mapgen stone with rockmech lithologies by depth layer (new chunks only).
rockmech_replace_stone (Replace mapgen stone) bool true

# Remove soil (dirt, sand, gravel, clay, snow) during map generation to expose bedrock.
# Usually set per world in <world>/rockmech.conf rather than globally.
rockmech_strip_soil (Strip soil) bool false

# Replace every other natural solid node (ores, permafrost, ice, biome stones, ...) with the
# lithology at that position during map generation, so only rockmech lithologies remain.
# Usually set per world in <world>/rockmech.conf.
rockmech_only_lithologies (Only lithologies) bool false

# Lithology distribution. layers: by depth (sandstone, shale, granite downward).
# zones: vertical zones along x, each zone_width wide (shale around x = 0, granite to the east,
# sandstone to the west, repeating). Usually set per world in <world>/rockmech.conf.
rockmech_lith_mode (Lithology distribution) enum layers layers,zones

# Width (m) of each lithology zone in zones mode.
rockmech_zone_width (Lithology zone width) int 80 4 10000

# Zone order for zones mode, comma separated (e.g. "granite" for a granite-only world). Empty = shale,granite,sandstone.
rockmech_zone_order (Lithology zone order) string

[Joint sets]

# Joint plane normals are snapped to an integer lattice of this density so that joint
# traces connect exactly between blocks. Larger = closer to the input orientation
# (about 1 degree at 8) but more texture variants. The effective orientation is shown
# by the compass. Changing it changes the joint geometry of the world.
rockmech_joint_lattice (Joint orientation lattice) int 8 2 16

# Override joint set 1: "dip,dipdir,spacing,persistence,phi"
# (degrees, degrees clockwise from north, m >= 1, 0-1, degrees).
# Empty = use the default in config.lua.
rockmech_joint_set_1 (Joint set 1 override) string

# Override joint set 2: "dip,dipdir,spacing,persistence,phi". Empty = default.
rockmech_joint_set_2 (Joint set 2 override) string

# Override joint set 3: "dip,dipdir,spacing,persistence,phi". Empty = default.
rockmech_joint_set_3 (Joint set 3 override) string

# Additional joint set 4: "dip,dipdir,spacing,persistence,phi". Empty = none. Sets must be consecutive.
rockmech_joint_set_4 (Joint set 4) string

# Additional joint set 5. Empty = none.
rockmech_joint_set_5 (Joint set 5) string

# Additional joint set 6. Empty = none.
rockmech_joint_set_6 (Joint set 6) string

[Rock mass strength]

# Model scale N (similitude): 1 m in game represents N m in reality.
# All strengths with stress units (sigma_ci, sigma_t, joint cohesion) are divided by N; friction angles are unchanged.
# Every failure mode then has the factor of safety of a real excavation N times larger.
rockmech_model_scale (Model scale N) float 1.0 0.01 1000

# Rock mass quality GSI for all lithologies (Hoek-Brown). "GSI" or "surface GSI,deep GSI,transition depth (m)".
# Depth is measured below surface_y. Represents only sub-grid fracturing (< 1 m) and size effect:
# the modelled joint sets are handled by joint slip, do not count them again. Empty = 100 (intact rock).
# A rock node with param2 = 1..100 uses that value as its GSI instead.
rockmech_gsi (GSI profile, all lithologies) string

# GSI profile for sandstone (overrides rockmech_gsi). Same format.
rockmech_gsi_sandstone (GSI profile: sandstone) string

# GSI profile for shale (overrides rockmech_gsi). Same format.
rockmech_gsi_shale (GSI profile: shale) string

# GSI profile for granite (overrides rockmech_gsi). Same format.
rockmech_gsi_granite (GSI profile: granite) string

# Hoek-Brown disturbance factor D (0 = undisturbed, 1 = heavy blast damage).
rockmech_rockmass_d (Disturbance factor D) float 0 0 1

[Deviatoric failure]

# Check exposed rock for deviatoric failure after excavation:
# sigma1 >= sigma3 + sigma_ci * (mb * sigma3 / sigma_ci + s)^a (Hoek-Brown rock mass, see GSI and model scale).
rockmech_failure_enabled (Enable deviatoric failure) bool true

# Radius (m) checked around each change made by the player (dig, place, blast): joint slip, deviatoric failure, key blocks.
rockmech_failure_radius (Check radius) int 6 1 32

# Cells removed by the physics (spalling, falls, sliding blocks) are checked again within radius * this factor.
rockmech_failure_secondary_factor (Secondary radius factor) float 0.5 0 1

# Half-width w of the horizontal load-spreading window (2w+1)^2.
# sigma_y = (sum of column loads in the window, incl. roof loads over openings) / (solid cells in window).
rockmech_failure_window (Load-spreading window half-width) int 4 1 16

# Unit weight (kN/m3) of non-lithology solid nodes (dirt, ores, ...) in the column load.
rockmech_failure_gamma_other (Unit weight of other solids) float 20 0 50

# Height (m) scanned upward for the column load; rock beyond is assumed continuous up to y0.
rockmech_failure_overburden_scan (Overburden scan height) int 64 0 512

# Initial in-situ stress (HUD / info page / compass): maximum height scanned upward to find the ground surface (m).
# Voids below the surface are counted as rock (pre-excavation state); beyond this height rock is assumed to continue to surface_y.
rockmech_failure_initial_scan_max (Initial stress scan height) int 512 16 4096

# Legacy: extra factor on sigma_ci. Prefer rockmech_model_scale, which also scales tensile strength and joint cohesion.
rockmech_failure_strength_scale (Strength scale, legacy) float 1.0 0.001 10

# Maximum number of cascade rounds after a failure.
rockmech_failure_max_cascade (Max cascade rounds) int 5 0 50

# Delay between cascade rounds (s).
rockmech_failure_cascade_delay (Cascade delay) float 0.6 0 10

# Log a warning when one check takes longer than this (ms).
rockmech_failure_budget_ms (Time budget per check) float 5 0.1 1000

[Warning cracks]

# Rock nodes around the player with 1 < FS < this value get a crack overlay (per-player display, /rockmech_fswarn).
rockmech_fswarn_max (Crack warning FS threshold) float 1.5 1.01 5

[Key blocks]

# Blocks completely bounded by joints and free faces are checked under gravity:
# fall, planar sliding or wedge sliding (joint c and phi from the joint plane of weakness settings).
rockmech_keyblock_enabled (Enable key blocks) bool true

# Blocks larger than this many nodes are treated as part of the rock mass (not checked).
rockmech_keyblock_max_cells (Max key block size) int 500 1 5000

# Cohesion (MPa) of the joints bounding a key block (open, persistent joints: usually 0). Friction angle is rockmech_weakness_phi.
rockmech_keyblock_c (Key block joint cohesion) float 0 0 10

# Stress clamping (Unwedge-style field stress with relaxation factor): each joint contact gets a normal force
# lambda * sigma_n * A from the local stress; faces that stay in contact while the block moves add friction.
# 0 = gravity only. Also applied to overhang groups cut off by joints (they stay if clamped or interlocked).
rockmech_keyblock_clamp (Key block stress clamping factor) float 0.5 0 2

# Stress clamping factor for mass (prism) sliding. 0 = stress near the slope face taken as relaxed (usual slope practice).
rockmech_keyblock_mass_clamp (Mass sliding stress clamping factor) float 0 0 2

# Mass sliding: the whole rock mass above a daylighting joint plane (planar) or a pair of planes (wedge)
# slides out together; internal joints carry no strength. Rock bridges count with the rock mass tensile strength.
rockmech_keyblock_mass (Mass sliding) bool true

# Read radius (m) for mass sliding; a mass reaching the edge is treated as not released.
rockmech_keyblock_mass_radius (Mass sliding radius) int 28 8 64

# Maximum cells in a sliding mass.
rockmech_keyblock_mass_max (Mass sliding max cells) int 8000 100 100000

# Maximum (joint combination, plane) candidates expanded per check.
rockmech_keyblock_mass_floods (Mass sliding candidates per check) int 48 1 200

[Induced failure only]

# Only failures caused by excavation: the generated terrain is taken as in equilibrium.
# Before/after comparison: only stable -> unstable transitions fail; a cell or overhang that already had FS <= 1 before the change is left alone
# (a pre-existing unstable overhang still falls once it is completely unsupported).
rockmech_induced_enabled (Only excavation-induced failure) bool true

# A pre-existing unstable overhang (FS < 1 before) still falls when the player digs it or right next to it,
# if it has at most this many nodes (larger ones are left alone to avoid huge collapses).
rockmech_induced_touch_max (Max size of a touched pre-existing overhang that falls) int 200 0 100000

[Joint plane of weakness]

# Slip along joints crossing an exposed cell (Jaeger single plane of weakness): tau >= c + sigma_n * tan(phi).
rockmech_weakness_enabled (Enable joint slip) bool true

# Joint cohesion c (MPa), shared by all joint sets.
rockmech_weakness_c (Joint cohesion) float 0.05 0 100

# Joint friction angle phi (degrees), shared by all joint sets.
rockmech_weakness_phi (Joint friction angle) float 30 0 89

# Kinematic check: a joint must dip toward an exposed side face (or the cell must be exposed below) to slip out.
rockmech_weakness_daylight (Require daylighting) bool true

# Lateral limit (deg) for planar sliding (Markland): the joint dip direction must be within this angle
# of the exposed face's outward normal, unless the cell is exposed on every side the dip direction points to
# (a convex corner with lateral release). Wedges on joint intersections are handled by key blocks.
rockmech_weakness_lateral_limit (Lateral limit for planar sliding) float 20 0 90

# Wedge sliding: where two joint sets both cross a cell, the cell may slide along their line of intersection
# (FS = (2c + (sigma_n1 + sigma_n2) tan phi) / shear along the line). The line must daylight like a plane does.
rockmech_weakness_wedge (Wedge sliding on joint intersections) bool true

# Cell-by-cell joint slip (Jaeger / wedge) only for joint sets with spacing <= this (m): the ubiquitous-joint idea.
# Wider, explicitly modelled joints are then handled only by block mechanics (key blocks, prisms, overhangs).
rockmech_weakness_max_spacing (Max spacing for cell-by-cell joint slip) float 100 0 100

[Stress solver prototype]

rockmech_fem_enabled (Stress solver prototype) bool true

# Let the stress solver decide collapses: after an excavation, exposed rock fails cell by cell in tension or shear,

rockmech_fem_radius (Solver radius) int 12 4 32

rockmech_fem_nu (Poisson ratio) float 0.25 0 0.45

# Conjugate gradient relative residual.
rockmech_fem_tol (Solver tolerance) float 0.001 0.000001 0.1

rockmech_fem_max_iter (Solver iteration limit) int 1500 10 20000

[Ground support]

# Shotcrete support pressure on the covered rock faces (MPa), applied as an equivalent

[Blasting]

rockmech_blast_enabled (Enable blast fragmentation) bool true

# Reference UCS (MPa) whose fragmentation radius equals the blast radius.
rockmech_blast_ucs_ref (Blast reference UCS) float 60 0.1 1000

# Fraction of the fragmentation radius inside which rock is thrown out.
rockmech_blast_inner (Blast ejection fraction) float 0.5 0 1

# Drill-and-blast: blast radius R0 of one emulsion cartridge (m); n cartridges in a hole give R0 * n^(1/3).
rockmech_blast_charge_radius (Cartridge blast radius) float 2.0 0.1 20

# Maximum number of cartridges in one drill hole.
rockmech_blast_max_cartridges (Cartridges per hole) int 4 1 20

# Hole depth (m): the charge detonates this deep behind the rock face (hole bottom).
rockmech_blast_hole_depth (Drill hole depth) int 2 1 10

# Radius factor of an unconfined charge laid on a rock face without a drill hole.
rockmech_blast_surface_factor (Unconfined charge factor) float 0.3 0 1

# Confinement: a drill hole whose only free face is its collar face has its radius multiplied by this.
rockmech_blast_confine_factor (Confinement factor) float 0.75 0 1

# Free-face bonus: when a free face (any non-rock cell, excluding the collar face) lies within the blast radius

rockmech_blast_free_face_k (Free-face bonus) float 1.0 0 10

rockmech_blast_detonator_delay (Blasting machine countdown) float 5 0 60

# ANFO column charge (vertical bench holes): blast radius of each 1.5 m segment (m).
rockmech_blast_anfo_radius (ANFO segment radius) float 2.5 0.1 10

# ANFO: stemming length left uncharged at the collar (m).
rockmech_blast_anfo_stemming (ANFO stemming) float 1.5 0 10

[Overhang]

# Overhanging rock (cantilevers, thin roofs) breaks at its root and falls when its distance from
# a supported rock column exceeds L_max = sqrt(sigma_t,mass * t / (3 * gamma)),
# sigma_t,mass = sigma_t * s * mi / mb (Hoek-Brown, see GSI) / model scale N,
# where t is the local rock thickness. Can also be toggled in game.
rockmech_overhang_enabled (Enable overhang breaking) bool true

# Legacy: extra factor on tensile strength. Prefer rockmech_model_scale.
rockmech_overhang_scale (Overhang tensile strength scale, legacy) float 1.0 0.001 100

# Tensile strength (MPa) of non-lithology bonded nodes (dirt, ores, ...).
rockmech_overhang_tensile_other (Tensile strength of other nodes) float 2.0 0 100

# Check radius (m). Overhangs reaching the edge of this region are left alone (conservative).
rockmech_overhang_radius (Overhang check radius) int 16 4 48

# Overhangs larger than this many nodes are left alone (conservative).
rockmech_overhang_max_voxels (Max overhang size) int 30000 1 200000

# Joints carry no tension across the faces they cut: the root section of a cantilever only counts intact faces
# (L_max scales with intact/total), and a block cut off completely by joints falls.
rockmech_overhang_joints (Joints cut overhangs) bool true

# Terzaghi rock load: loosened height Hp = k * (B + Ht), B = 2 * hydraulic radius, Ht = opening height.
# Roof load height = min(thickness, Hp); a rock bolt must be at least that long to count.
# k follows GSI: 0 at or above GSI a, k_b at GSI b, k_c at or below GSI c, linear in between.
rockmech_overhang_terzaghi_gsi_a (Terzaghi: GSI with k = 0) float 85 0 100
rockmech_overhang_terzaghi_gsi_b (Terzaghi: middle GSI) float 50 0 100
rockmech_overhang_terzaghi_k_b (Terzaghi: k at middle GSI) float 0.35 0 10
rockmech_overhang_terzaghi_gsi_c (Terzaghi: low GSI) float 25 0 100
rockmech_overhang_terzaghi_k_c (Terzaghi: k at low GSI) float 1.1 0 10

[Shear of hanging roofs]

# A hanging roof segment is also checked in shear along its supported perimeter, with the same load as bending:
# V = (gamma * h - p) * HR per metre of perimeter, R = k * sigma_t * t_eff. The lower of the two FS governs.
rockmech_shear_enabled (Shear check of hanging roofs) bool true

# tau0 = k * sigma_t (Griffith k = 2).
rockmech_shear_k (Shear strength factor k) float 2 0 20

# How far (m) roof load searches for the supporting rock when it is passed on to the pillars.
rockmech_shear_max_span (Roof load transfer span) int 32 1 64
